1. Create `packages/dashboard/src/lib/api.ts` — typed fetch wrappers:

   Define response types (duplicated from core types, simplified for client):
   ```typescript
   export interface DashboardStats {
     pending: number
     entities: number
     recentEpisodes: Array<{
       id: string; session_id: string; agent_id: string;
       event_type: string; created_at: string
     }>
   }

   export interface ApprovalItem {
     id: string; item_type: string; item_id: string; status: string;
     reason: string | null; created_at: string; resolved_at: string | null;
     metadata: {
       fact: {
         entity_name: string; entity_type: string; observation: string;
         confidence: number; evidence_quote: string;
         related_entities: Array<{ name: string; type: string; relation_type: string }>
       };
       source_episode_ids?: string[];
       merge_candidate_ids?: string[];
     } | null;
   }

   export interface GraphData {
     nodes: Array<{ id: string; name: string; type: string; val: number }>
     links: Array<{ source: string; target: string; type: string }>
   }

   export interface EntityDetail {
     entity: { id: string; name: string; type: string; confidence: number; created_at: string; updated_at: string }
     observations: Array<{ id: string; content: string; confidence: number; created_at: string }>
     connected: Array<{ id: string; name: string; type: string; relation_type: string }>
   }
   ```

   Export fetch functions:
   - `fetchDashboard(): Promise<DashboardStats>` — GET `/api/dashboard`
   - `fetchApprovals(): Promise<{ items: ApprovalItem[] }>` — GET `/api/approvals`
   - `resolveApproval(id: string, status: 'approved' | 'rejected', edited_content?: string): Promise<{ status: string; id: string }>` — PATCH `/api/approvals/${id}`
   - `fetchGraph(): Promise<GraphData>` — GET `/api/graph`
   - `fetchEntityDetail(id: string): Promise<EntityDetail>` — GET `/api/entities/${id}`
   - `fetchEpisodes(limit?: number): Promise<{ episodes: Array<...> }>` — GET `/api/episodes?limit=${limit}`

   Each function: `const res = await fetch(url, opts); if (!res.ok) throw new Error(...); return res.json()`
