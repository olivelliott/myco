```typescript
import { Ollama } from 'ollama';

const EMBED_MODEL = 'nomic-embed-text';
const HEALTH_TIMEOUT_MS = 2000;

let client: Ollama | null = null;

function getClient(): Ollama {
  if (!client) {
    client = new Ollama({ host: process.env.OLLAMA_HOST ?? 'http://127.0.0.1:11434' });
  }
  return client;
}

/**
 * Generate an embedding vector for the given text using Ollama.
 * Returns null if Ollama is unavailable (caller handles graceful degradation).
 */
export async function embedText(text: string): Promise<number[] | null> {
  try {
    const ollama = getClient();
    const response = await ollama.embed({
      model: EMBED_MODEL,
      input: text,
      options: {},
    });
    return response.embeddings[0] ?? null;
  } catch {
    // Ollama unavailable — caller handles null (SRCH-04 graceful degradation)
    return null;
  }
}
```

Note: AbortSignal.timeout is handled by the Ollama client options or HTTP-level timeout. The 2s timeout applies at the health-check level. If the embed call itself takes longer but succeeds, that is acceptable for inline embedding.
